SELECT region, SUM(revenue) AS total_revenue
FROM sales_data
WHERE date >= DATE_TRUNC('month', CURRENT_DATE - INTERVAL '1 month')
  AND date < DATE_TRUNC('month', CURRENT_DATE)
GROUP BY region
ORDER BY total_revenue DESC;
